# 2️⃣  Validation Filter
# ✅ Fix: كل entity لازم تظهر في relationships
# ══════════════════════════════════════════════════════════
def is_valid_example(answer: str) -> bool:
    """
    تتأكد إن:
    1. الـ answer فيها Solution 1 و Solution 2
    2. كل entity موجودة في الـ relationships section
    """
    if "## Solution 1" not in answer or "## Solution 2" not in answer:
        return False

    # استخرج كل entities
    entities = re.findall(r"####\s+(\w+)[:\s]", answer)
    if not entities:
        return False

    # استخرج الـ relationships sections
    rel_section = ""
    if "### Relationships" in answer:
        parts = answer.split("### Relationships")
        for part in parts[1:]:
            rel_section += part

    if not rel_section:
        return False

    # تأكد إن كل entity موجودة في الـ relationships
    skip_entities = {"Transaction", "Transaction1", "Transaction2", "Transaction3"}
    for entity in entities:
        if entity in skip_entities:
            continue
        if entity not in rel_section:
            return False

    return True


def validate_dataset(examples, label=""):
    valid   = [ex for ex in examples if is_valid_example(ex["answer"])]
    invalid = len(examples) - len(valid)
    print(f"   {label}: {len(examples)} total → {len(valid)} valid, {invalid} removed")
    return valid

# ✅ Fix: بنغير السؤال بس — الـ answer بيفضل ثابت

def get_domain(q):
    patterns = [
        ("construction enterprise", "construction"),
        ("sales representatives",   "sales"),
        ("football tournament",     "football"),
        ("EGYPTAIR",                "airline"),
        ("education system",        "education"),
        ("COMPANY database",        "company"),
        ("Hotel Booking",           "hotel"),
        ("Transportation Fleet",    "fleet"),
        ("Restaurant",              "restaurant"),
        ("hospital",                "hospital"),
        ("university",              "university"),
        ("bank",                    "bank"),
        ("airline company",         "airline"),
        ("Library Management",      "library"),
        ("Online Shopping",         "shopping"),
        ("healthcare clinic",       "clinic"),
        ("research",                "research"),
    ]
    for phrase, domain in patterns:
        if phrase.lower() in q.lower():
            return domain
    return None


DOMAIN_POOL = [
    "hospital", "university", "airline", "bank", "hotel", "restaurant",
    "school", "pharmacy", "clinic", "military", "logistics", "insurance",
    "library", "retail", "real estate", "construction", "company",
    "government", "manufacturing", "telecom", "e-commerce", "healthcare",
]

EXTRA_ATTRS = [
    "Email", "Phone", "Address", "Status", "CreatedAt", "UpdatedAt",
    "Notes", "Rating", "Category", "Priority", "Cost", "Description",
]

CARD_PHRASES = {
    "only one area":              "more than one area",
    "more than one area":         "only one area",
    "one prerequisite course":    "more than one prerequisite course",
    "more than one prerequisite": "one prerequisite",
    "only one teacher":           "more than one teacher",
}

Q_TEMPLATES = [
    "Design an ER model for a {domain} system where {rest}",
    "A {domain} organization needs a database. {rest}",
    "Build a complete database schema for a {domain}. {rest}",
    "Describe entities and relationships for a {domain} database. {rest}",
    "What entities and relationships exist in a {domain} management system? {rest}",
    "For a {domain} application, identify all entities and relationships. {rest}",
    "Create an ER diagram for a {domain} where {rest}",
    "Model a {domain} database with proper entities and relationships. {rest}",
    "A professor asked: design a database for a {domain}. {rest}",
    "Give me the ER model for a {domain} system. {rest}",
    "Suppose you are designing a {domain} database. {rest}",
    "A {domain} company wants to track: {rest}",
]


def make_key(q):
    return hashlib.md5(q.strip().encode()).hexdigest()


def augment_to_target(all_source, eval_raw, train_raw, target=1500):
    eval_keys = {make_key(ex["question"]) for ex in eval_raw}
    seen      = set(eval_keys)
    aug_data  = []

    def add(q, a):
        # ✅ Fix: validate قبل الإضافة
        if not is_valid_example(a):
            return False
        k = make_key(q)
        if k not in seen:
            seen.add(k)
            aug_data.append({"question": q, "answer": a})
            return True
        return False

    # أضف الـ original train أولاً
    for ex in train_raw:
        add(ex["question"], ex["answer"])

    print(f"\n🔄 Augmentation started — original train: {len(aug_data)}, target: {target}")
    print("=" * 60)

    # ──────────────────────────────────────────────────────
    # S1: Question Paraphrase — answer ثابت ✅
    # ──────────────────────────────────────────────────────
    for ex in all_source:
        domain = get_domain(ex["question"]) or "system"
        sentences = ex["question"].split(".")
        rest = ".".join(sentences[1:]).strip() if len(sentences) > 1 else ex["question"]
        for tmpl in Q_TEMPLATES:
            new_q = tmpl.format(domain=domain, rest=rest)
            add(new_q, ex["answer"])   # ✅ answer ثابت
    print(f"  ✅ S1 Question Paraphrase   → total: {len(aug_data)}")

    # ──────────────────────────────────────────────────────
    # S2: Extra Attribute in Question only — answer ثابت ✅
    # ──────────────────────────────────────────────────────
    for ex in all_source:
        for attr in EXTRA_ATTRS:
            new_q = ex["question"].rstrip() + f"\n- Each entity must also record a {attr}."
            add(new_q, ex["answer"])   # ✅ answer ثابت
    print(f"  ✅ S2 Attribute in Question → total: {len(aug_data)}")

    # ──────────────────────────────────────────────────────
    # S3: Cardinality Flip in Question only — answer ثابت ✅
    # ──────────────────────────────────────────────────────
    for ex in all_source:
        for old_p, new_p in CARD_PHRASES.items():
            if old_p.lower() in ex["question"].lower():
                new_q = re.sub(re.escape(old_p), new_p, ex["question"], flags=re.IGNORECASE)
                if new_q != ex["question"]:
                    add(new_q, ex["answer"])   # ✅ answer ثابت
    print(f"  ✅ S3 Cardinality Flip      → total: {len(aug_data)}")

    # ──────────────────────────────────────────────────────
    # S4: Solution Focus — question بيتحدد الـ solution المطلوبة
    # ──────────────────────────────────────────────────────
    for ex in all_source:
        if "## Solution 2" in ex["answer"]:
            q2 = ex["question"].rstrip() + "\n(Provide Solution 2 with Transaction entity only)"
            a2 = "## Solution 2" + ex["answer"].split("## Solution 2")[1]
            add(q2, a2)
        if "## Solution 1" in ex["answer"]:
            q1 = ex["question"].rstrip() + "\n(Provide Solution 1 with direct M:M relationship only)"
            part = ex["answer"].split("## Solution 2")[0] if "## Solution 2" in ex["answer"] else ex["answer"]
            a1 = "## Solution 1" + part.split("## Solution 1")[1] if "## Solution 1" in part else part
            add(q1, a1)
    print(f"  ✅ S4 Solution Focus        → total: {len(aug_data)}")

    # ──────────────────────────────────────────────────────
    # S5: Constraint Additions in Question only — answer ثابت ✅
    # ──────────────────────────────────────────────────────
    if len(aug_data) < target:
        CONSTRAINTS = [
            "Each entity must have a unique identifier.",
            "All dates must be stored in ISO format.",
            "Foreign keys must be indexed.",
            "No NULL values allowed in primary keys.",
            "All monetary values stored as DECIMAL(10,2).",
        ]
        for ex in all_source:
            for c in CONSTRAINTS:
                new_q = ex["question"].rstrip() + f"\nConstraint: {c}"
                add(new_q, ex["answer"])   # ✅ answer ثابت
        print(f"  ✅ S5 Constraints           → total: {len(aug_data)}")
    else:
        print(f"  ⏭️  S5 skipped — target reached")

    # ──────────────────────────────────────────────────────
    # S6: Attribute Pairs in Question only — answer ثابت ✅
    # ──────────────────────────────────────────────────────
    if len(aug_data) < target:
        attr_pairs = list(itertools.combinations(EXTRA_ATTRS[:6], 2))
        for ex in all_source:
            for a1_attr, a2_attr in attr_pairs[:10]:
                new_q = ex["question"].rstrip() + f"\n- Track {a1_attr} and {a2_attr} for all entities."
                add(new_q, ex["answer"])   # ✅ answer ثابت
        print(f"  ✅ S6 Attribute Pairs       → total: {len(aug_data)}")
    else:
        print(f"  ⏭️  S6 skipped — target reached")

    # ──────────────────────────────────────────────────────
    # S7: Domain Pool Injection في السؤال بس — answer ثابت ✅
    # ──────────────────────────────────────────────────────
    if len(aug_data) < target:
        for nd in DOMAIN_POOL:
            for ex in all_source:
                d1 = get_domain(ex["question"])
                if not d1 or d1.lower() == nd.lower():
                    continue
                # غير السؤال بس
                new_q = re.sub(
                    r"\b" + re.escape(d1) + r"\b", nd,
                    ex["question"], flags=re.IGNORECASE
                )
                if new_q != ex["question"]:
                    add(new_q, ex["answer"])   # ✅ answer ثابت
        print(f"  ✅ S7 Domain Injection      → total: {len(aug_data)}")
    else:
        print(f"  ⏭️  S7 skipped — target reached")

    print("=" * 60)
    print(f"  📊 Final size  : {len(aug_data)}")
    print(f"  📊 Multiplier  : {len(aug_data) / max(len(train_raw), 1):.1f}x")
    status = "✅ REACHED" if len(aug_data) >= target else f"⚠️  short by {target - len(aug_data)}"
    print(f"  📊 Target {target}: {status}")
    return aug_data

def format_llama3(example):
    return {"text": (
        "<|begin_of_text|>"
        "<|start_header_id|>system<|end_header_id|>\n"
        f"{SYSTEM_PROMPT}<|eot_id|>"
        "<|start_header_id|>user<|end_header_id|>\n"
        f"{example['question']}<|eot_id|>"
        "<|start_header_id|>assistant<|end_header_id|>\n"
        f"{example['answer']}<|eot_id|>"
    )}


train_dataset = train_dataset.map(format_llama3)
eval_dataset  = eval_dataset.map(format_llama3)

# 8️⃣  Overfitting Detector Callback
# ══════════════════════════════════════════════════════════
class OverfitDetectorCallback(TrainerCallback):
    def __init__(self, gap_threshold=0.5, patience=3):
        self.gap_threshold = gap_threshold
        self.patience      = patience
        self.history       = []
        self.best_eval     = float("inf")
        self.bad_steps     = 0

    def on_evaluate(self, args, state, control, metrics=None, **kwargs):
        if not metrics:
            return
        eval_loss  = metrics.get("eval_loss")
        train_loss = None
        for log in reversed(state.log_history):
            if "loss" in log and "eval_loss" not in log:
                train_loss = log["loss"]
                break
        if eval_loss is None or train_loss is None:
            return

        gap = eval_loss - train_loss
        self.history.append((state.global_step, train_loss, eval_loss))

        print(f"\n{'='*55}")
        print(f"  📊 Step {state.global_step}")
        print(f"     Train : {train_loss:.4f} | Eval : {eval_loss:.4f} | Gap : {gap:+.4f}")

        if eval_loss < self.best_eval:
            self.best_eval = eval_loss
            self.bad_steps = 0
            print(f"  ✅ New best eval loss: {self.best_eval:.4f}")
        else:
            self.bad_steps += 1
            print(f"  ⚠️  No improvement ({self.bad_steps}/{self.patience})")

        status = "🚨 OVERFIT" if gap > self.gap_threshold else "✅ Gap OK"
        print(f"  {status} (gap={gap:+.4f}, threshold={self.gap_threshold})")
        print(f"{'='*55}\n")

    def on_train_end(self, args, state, control, **kwargs):
        if not self.history:
            return
        print("\n" + "=" * 60)
        print(f"  {'Step':>6}  {'Train':>8}  {'Eval':>8}  {'Gap':>8}  Status")
        print("  " + "-" * 54)
        for step, tl, el in self.history:
            gap  = el - tl
            flag = "🚨" if gap > self.gap_threshold else ("⚠️" if gap > 0.2 else "✅")
            print(f"  {step:>6}  {tl:>8.4f}  {el:>8.4f}  {gap:>+8.4f}  {flag}")
        print("=" * 60)

# ══════════════════════════════════════════════════════════

training_args = TrainingArguments(
    output_dir="./llama3_8B_lora_v4",

    # Batch
    per_device_train_batch_size=1,
    per_device_eval_batch_size=1,
    gradient_accumulation_steps=4,    # effective batch = 4

    max_steps=1000,                   # ✅ Fix: كان 500

    # LR

    lr_scheduler_type="cosine",
    warmup_ratio=0.1,                 # ✅ Fix: بدل warmup_steps ثابتة

    # Regularization
    weight_decay=0.05,                # ✅ Fix: كان 0.01

    # Precision
    fp16=True,

    # Logging

    # Eval & Save
    eval_strategy="steps",
    eval_steps=eval_steps,
    save_strategy="steps",
    save_steps=eval_steps,
    save_total_limit=3,

    # Best model
    load_best_model_at_end=True,
    metric_for_best_model="eval_loss",
    greater_is_better=False,

    # Optimizer
    optim="paged_adamw_8bit",

    report_to="none",

)
