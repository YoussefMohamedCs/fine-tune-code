from datasets import load_dataset, Dataset
from transformers import (
    AutoModelForCausalLM,
    AutoTokenizer,
    TrainingArguments,
    Trainer,
    DataCollatorForLanguageModeling,
    BitsAndBytesConfig,
    TrainerCallback,
    EarlyStoppingCallback,
)
from peft import LoraConfig, get_peft_model

# ══════════════════════════════════════════════════════════
# 1️⃣  Load Dataset
# ══════════════════════════════════════════════════════════
dataset = load_dataset(
    "json",
    data_files="/kaggle/input/datasets/tagprogaramming/18data/er_database_qa.json"
)
raw_examples = dataset["train"].to_list()
print(f"✅ Loaded {len(raw_examples)} examples")

random.seed(42)
random.shuffle(raw_examples)

eval_size = max(2, int(len(raw_examples) * 0.15))
eval_raw  = raw_examples[:eval_size]
train_raw = raw_examples[eval_size:]
print(f"   Train: {len(train_raw)} | Eval: {len(eval_raw)}")

# ══════════════════════════════════════════════════════════
# 2️⃣  Augmentation
# ══════════════════════════════════════════════════════════

def get_domain(q):
    patterns = [
        ("construction enterprise", "construction"),
        ("sales representatives",   "sales"),
        ("football tournament",     "football"),
        ("EGYPTAIR",                "airline"),
        ("education system",        "education"),
        ("COMPANY database",        "company"),
        ("Hotel Booking",           "hotel"),
        ("Transportation Fleet",    "fleet"),
        ("Restaurant",              "restaurant"),
        ("hospital",                "hospital"),
        ("university",              "university"),
        ("bank",                    "bank"),
        ("airline company",         "airline"),
        ("Library Management",      "library"),
    ]
    for phrase, domain in patterns:
        if phrase.lower() in q.lower():
            return domain
    return None

DOMAIN_POOL = [
    "hospital","university","airline","bank","hotel","restaurant",
    "school","pharmacy","clinic","military","logistics","insurance",
    "library","retail","real estate","construction","company",
    "government","manufacturing","telecom","e-commerce","healthcare",
]

EXTRA_ATTRS = [
    "Email","Phone","Address","Status","CreatedAt","UpdatedAt",
    "Notes","Rating","Category","Priority","Cost","Description",
]

CARD_PHRASES = {
    "only one area":              "more than one area",
    "more than one area":         "only one area",
    "one prerequisite course":    "more than one prerequisite course",
    "more than one prerequisite": "one prerequisite",
    "only one teacher":           "more than one teacher",
}

Q_TEMPLATES = [
    "Design an ER model for a {domain} system where {rest}",
    "A {domain} organization needs a database. {rest}",
    "Build a complete database schema for a {domain}. {rest}",
    "Describe entities and relationships for a {domain} database. {rest}",
    "What entities and relationships exist in a {domain} management system? {rest}",
    "For a {domain} application, identify all entities and relationships. {rest}",
    "Create an ER diagram for a {domain} where {rest}",
    "Model a {domain} database with proper entities and relationships. {rest}",
    "A professor asked: design a database for a {domain}. {rest}",
    "Give me the ER model for a {domain} system. {rest}",
    "Suppose you are designing a {domain} database. {rest}",
    "A {domain} company wants to track: {rest}",
]

def make_key(q):
    return hashlib.md5(q.strip().encode()).hexdigest()


def augment_to_target(all_source, eval_raw, train_raw, target=1500):
    eval_keys = {make_key(ex["question"]) for ex in eval_raw}
    seen      = set(eval_keys)
    aug_data  = []

    def add(q, a):
        k = make_key(q)
        if k not in seen:
            seen.add(k)
            aug_data.append({"question": q, "answer": a})
            return True
        return False

    for ex in train_raw:
        add(ex["question"], ex["answer"])

    print(f"\n🔄 Augmentation started — original train: {len(aug_data)}, target: {target}")
    print("="*60)

    # S1: Entity Substitution
    for ex1 in all_source:
        d1 = get_domain(ex1["question"])
        if not d1: continue
        for ex2 in all_source:
            if ex1 is ex2: continue
            d2 = get_domain(ex2["question"])
            if not d2 or d1 == d2: continue
            new_q = re.sub(r"\b"+re.escape(d1)+r"\b", d2, ex1["question"], flags=re.IGNORECASE)
            new_a = re.sub(r"\b"+re.escape(d1)+r"\b", d2, ex1["answer"],   flags=re.IGNORECASE)
            if new_q != ex1["question"]:
                add(new_q, new_a)
    print(f"  ✅ S1 Entity Substitution   → total: {len(aug_data)}")

    # S2: Domain Pool Injection
    for nd in DOMAIN_POOL:
        for ex in all_source:
            d1 = get_domain(ex["question"])
            if not d1 or d1.lower() == nd.lower(): continue
            new_q = re.sub(r"\b"+re.escape(d1)+r"\b", nd, ex["question"], flags=re.IGNORECASE)
            new_a = re.sub(r"\b"+re.escape(d1)+r"\b", nd, ex["answer"],   flags=re.IGNORECASE)
            if new_q != ex["question"]:
                add(new_q, new_a)
    print(f"  ✅ S2 Domain Pool Injection → total: {len(aug_data)}")

    # S3: Single Attribute Addition
    for ex in all_source:
        for attr in EXTRA_ATTRS:
            new_q = ex["question"].rstrip() + f"\n- Each entity must also record a {attr}."
            new_a = re.sub(
                r"(\*\* \(Primary Key\)\n)",
                r"\1- **" + attr + r"**\n",
                ex["answer"], count=1
            )
            add(new_q, new_a)
    print(f"  ✅ S3 Attribute Addition    → total: {len(aug_data)}")

    # S4: Cardinality Flip
    for ex in all_source:
        for old_p, new_p in CARD_PHRASES.items():
            if old_p.lower() in ex["question"].lower():
                new_q = re.sub(re.escape(old_p), new_p, ex["question"], flags=re.IGNORECASE)
                if new_q != ex["question"]:
                    add(new_q, ex["answer"])
    print(f"  ✅ S4 Cardinality Flip      → total: {len(aug_data)}")

    # S5: Question Paraphrase
    for ex in all_source:
        domain = get_domain(ex["question"]) or "system"
        sentences = ex["question"].split(".")
        rest = ".".join(sentences[1:]).strip() if len(sentences) > 1 else ex["question"]
        for tmpl in Q_TEMPLATES:
            new_q = tmpl.format(domain=domain, rest=rest)
            add(new_q, ex["answer"])
    print(f"  ✅ S5 Question Paraphrase   → total: {len(aug_data)}")

    # S6: Solution Focus
    for ex in all_source:
        if "## Solution 2" in ex["answer"]:
            q2 = ex["question"].rstrip() + "\n(Provide Solution 2 with Transaction entity only)"
            a2 = "## Solution 2" + ex["answer"].split("## Solution 2")[1]
            add(q2, a2)
        if "## Solution 1" in ex["answer"]:
            q1 = ex["question"].rstrip() + "\n(Provide Solution 1 with direct M:M relationship only)"
            part = ex["answer"].split("## Solution 2")[0] if "## Solution 2" in ex["answer"] else ex["answer"]
            a1 = "## Solution 1" + part.split("## Solution 1")[1] if "## Solution 1" in part else part
            add(q1, a1)
    print(f"  ✅ S6 Solution Focus        → total: {len(aug_data)}")

    # S7: Attribute Pairs
    if len(aug_data) < target:
        attr_pairs = list(itertools.combinations(EXTRA_ATTRS[:6], 2))
        for ex in all_source:
            for a1_attr, a2_attr in attr_pairs[:10]:
                new_q = ex["question"].rstrip() + f"\n- Track {a1_attr} and {a2_attr} for all entities."
                add(new_q, ex["answer"])
        print(f"  ✅ S7 Attribute Pairs       → total: {len(aug_data)}")
    else:
        print(f"  ⏭️  S7 skipped — target reached")

    # S8: Constraint Additions
    if len(aug_data) < target:
        CONSTRAINTS = [
            "Each entity must have a unique identifier.",
            "All dates must be stored in ISO format.",
            "Foreign keys must be indexed.",
            "No NULL values allowed in primary keys.",
            "All monetary values stored as DECIMAL(10,2).",
        ]
        for ex in all_source:
            for c in CONSTRAINTS:
                new_q = ex["question"].rstrip() + f"\nConstraint: {c}"
                add(new_q, ex["answer"])
        print(f"  ✅ S8 Constraints           → total: {len(aug_data)}")
    else:
        print(f"  ⏭️  S8 skipped — target reached")

    # S9: Domain + Attribute combo
    if len(aug_data) < target:
        for nd in DOMAIN_POOL[:8]:
            for ex in all_source:
                d1 = get_domain(ex["question"])
                if not d1 or d1.lower() == nd.lower(): continue
                for attr in EXTRA_ATTRS[:4]:
                    new_q = re.sub(r"\b"+re.escape(d1)+r"\b", nd, ex["question"], flags=re.IGNORECASE)
                    new_q = new_q.rstrip() + f"\n- Also store {attr} for each entity."
                    new_a = re.sub(r"\b"+re.escape(d1)+r"\b", nd, ex["answer"], flags=re.IGNORECASE)
                    if new_q != ex["question"]:
                        add(new_q, new_a)
        print(f"  ✅ S9 Domain+Attr Combo     → total: {len(aug_data)}")
    else:
        print(f"  ⏭️  S9 skipped — target reached")

    print("="*60)
    print(f"  📊 Final size  : {len(aug_data)}")
    print(f"  📊 Multiplier  : {len(aug_data)/len(train_raw):.1f}x")
    status = "✅ REACHED" if len(aug_data) >= target else f"⚠️  short by {target - len(aug_data)}"
    print(f"  📊 Target {target}: {status}")
    return aug_data


augmented_train = augment_to_target(raw_examples, eval_raw, train_raw, target=1500)
random.shuffle(augmented_train)

EFFECTIVE_BATCH = 8
NUM_EPOCHS      = 5
steps_per_epoch = max(1, len(augmented_train) // EFFECTIVE_BATCH)
max_steps       = steps_per_epoch * NUM_EPOCHS
eval_steps      = steps_per_epoch
warmup_steps    = max(20, steps_per_epoch)

print(f"\n📐 Augmented size  : {len(augmented_train)}")
print(f"   Steps per epoch : {steps_per_epoch}")
print(f"   Max steps       : {max_steps}")
print(f"   Eval every      : {eval_steps} steps")
print(f"   Warmup steps    : {warmup_steps}")

train_dataset = Dataset.from_list(augmented_train)
eval_dataset  = Dataset.from_list(eval_raw)

# ══════════════════════════════════════════════════════════
# 3️⃣  Model & Tokenizer
# 🔧 v3: 4bit بدل 8bit عشان يتناسب مع 14GB ويسمح بـ max_length=2048
# ══════════════════════════════════════════════════════════
model_name = "meta-llama/Meta-Llama-3-8B-Instruct"

bnb_config = BitsAndBytesConfig(
    load_in_4bit=True,                       # ✅ v3: 4bit بدل 8bit
    bnb_4bit_compute_dtype=torch.float16,    # ✅ v3: أسرع وأقل memory
    bnb_4bit_use_double_quant=True,          # ✅ v3: double quant لتوفير memory أكتر
    bnb_4bit_quant_type="nf4",              # ✅ v3: أفضل accuracy مع 4bit
)

tokenizer = AutoTokenizer.from_pretrained(model_name, trust_remote_code=True)
tokenizer.pad_token = tokenizer.eos_token

model = AutoModelForCausalLM.from_pretrained(
    model_name,
    quantization_config=bnb_config,
    device_map="auto",
)

# ══════════════════════════════════════════════════════════
# 4️⃣  Format — Llama-3 Chat Template
# ══════════════════════════════════════════════════════════
SYSTEM_PROMPT = (
    "You are an ER modeling expert. No thinking tags. No prose. "
    "Always give exactly TWO solutions with entity types and relationships only: "
    "Solution 1: Direct modeling (if repetition is not needed) - use M:M relationship directly. "
    "Solution 2: MANDATORY Transaction entity (if repetition is needed) - ALWAYS resolve every M:M "
    "using a Transaction entity. Never skip Solution 2. Do not repeat the answer.\n\n"

    "=== EXAMPLE 1 ===\n"
    "Question: A database is decided for a large construction enterprise to support its resources "
    "management. Such an enterprise has a set of Departments and Projects. Each department has a set "
    "of Equipment. Equipment's are assigned to projects for a specific period of time...\n\n"
    "Answer:\n"
    "## Solution 1: Using M:M Relationship (Direct Assignment)\n"
    "### Entity Types\n"
    "#### Department: Department_Number (PK), Name (Unique), Budget\n"
    "#### Equipment: Equipment_Number (PK), Name, Location\n"
    "#### Project: Project_Number (PK), Name, Location\n"
    "### Relationships\n"
    "#### Owns: Department → Equipment | 1:M\n"
    "#### Assigned_To: Equipment ↔ Project | M:M | Attributes: Assignment_Start_Date, Assignment_Finish_Date\n\n"
    "## Solution 2: Resolving M:M Using Transaction Entity\n"
    "### Entity Types\n"
    "#### Department: Department_Number (PK), Name (Unique), Budget\n"
    "#### Equipment: Equipment_Number (PK), Name, Location, Department_Number (FK)\n"
    "#### Project: Project_Number (PK), Name, Location\n"
    "#### Transaction: Transaction_Number (PK), Assignment_Start_Date, Assignment_Finish_Date, "

    "### Relationships\n"
    "#### Owns: Department → Equipment | 1:M\n"
    "#### Equipment_Assignment: Equipment → Transaction | 1:M\n"
    "#### Project_Assignment: Project → Transaction | 1:M\n\n"

    "=== EXAMPLE 2 ===\n"
    "Question: A database contains information concerning sales representatives, sales areas, "

    "Answer:\n"
    "## Solution 1: Using M:M Relationship (Direct Modeling)\n"
    "### Entity Types\n"
    "#### Sales_Representative: Representative_Number (PK), Representative_Name (Unique), Sales_Commission\n"
    "#### Sales_Area: Area_Number (PK), Location, Commission_Rate\n"
    "#### Product: Product_Number (PK), Product_Name, Price\n"
    "### Relationships\n"
    "#### Sells: Sales_Representative ↔ Product ↔ Sales_Area | M:M:M | Attribute: Quantity_Sold\n\n"
    "## Solution 2: Resolving M:M Using Transaction Entity\n"
    "### Entity Types\n"
    "#### Sales_Representative: Representative_Number (PK), Representative_Name (Unique), "
    "Sales_Commission, Area_Number (FK)\n"
    "#### Sales_Area: Area_Number (PK), Location, Commission_Rate\n"
    "#### Product: Product_Number (PK), Product_Name, Price\n"
    "#### Transaction: Transaction_Number (PK), Sub_Quantity_Sold, Representative_Number (FK), "
    "Product_Number (FK), SalesArea_Number (FK)\n"
    "### Relationships\n"
    "#### Sales_Area_Transaction: Sales_Area → Transaction | 1:M\n"
    "#### Representative_Transaction: Sales_Representative → Transaction | 1:M\n"
    "#### Product_Transaction: Product → Transaction | 1:M\n\n"

    "=== NOW ANSWER THE FOLLOWING ==="
)


def format_llama3(example):
    return {"text": (
        "<|begin_of_text|>"
        "<|start_header_id|>system<|end_header_id|>\n"
        f"{SYSTEM_PROMPT}<|eot_id|>"
        "<|start_header_id|>user<|end_header_id|>\n"
        f"{example['question']}<|eot_id|>"
        "<|start_header_id|>assistant<|end_header_id|>\n"
        f"{example['answer']}<|eot_id|>"
    )}


train_dataset = train_dataset.map(format_llama3)
eval_dataset  = eval_dataset.map(format_llama3)

# ══════════════════════════════════════════════════════════
# 5️⃣  Tokenization
# 🔧 v3: max_length 768 → 2048 (الإصلاح الرئيسي!)
# الأطول Q+A في الداتاسيت ~1421 token + system prompt ~600 = ~2000
# ══════════════════════════════════════════════════════════
def tokenize(examples):
    return tokenizer(
        examples["text"],
        truncation=True,
        max_length=2048,   # ✅ v3: كان 768 → 2048 (السبب الرئيسي للإجابات الغلط!)
        padding="max_length",
    )

tokenized_train = train_dataset.map(tokenize, batched=True)
tokenized_eval  = eval_dataset.map(tokenize, batched=True)

# ══════════════════════════════════════════════════════════
# 6️⃣  LoRA
# ══════════════════════════════════════════════════════════
model.gradient_checkpointing_enable()
model.enable_input_require_grads()

lora_config = LoraConfig(
    r=32,
    lora_alpha=64,
    target_modules=["q_proj","v_proj","k_proj","o_proj","up_proj","down_proj","gate_proj"],
    lora_dropout=0.1,
    bias="none",
    task_type="CAUSAL_LM",
)
model = get_peft_model(model, lora_config)
model.print_trainable_parameters()

# ══════════════════════════════════════════════════════════
# 7️⃣  Overfitting Detector
# ══════════════════════════════════════════════════════════
class OverfitDetectorCallback(TrainerCallback):
    def __init__(self, gap_threshold=0.5, patience=3):
        self.gap_threshold = gap_threshold
        self.patience      = patience
        self.history       = []
        self.best_eval     = float("inf")
        self.bad_steps     = 0

    def on_evaluate(self, args, state, control, metrics=None, **kwargs):
        if not metrics: return
        eval_loss = metrics.get("eval_loss")
        train_loss = None
        for log in reversed(state.log_history):
            if "loss" in log and "eval_loss" not in log:
                train_loss = log["loss"]; break
        if eval_loss is None or train_loss is None: return
        gap = eval_loss - train_loss
        self.history.append((state.global_step, train_loss, eval_loss))
        print(f"\n{'='*55}")
        print(f"  📊 Step {state.global_step}")
        print(f"     Train : {train_loss:.4f} | Eval : {eval_loss:.4f} | Gap : {gap:+.4f}")
        if eval_loss < self.best_eval:
            self.best_eval = eval_loss; self.bad_steps = 0
            print(f"  ✅ New best: {self.best_eval:.4f}")
        else:
            self.bad_steps += 1
            print(f"  ⚠️  No improvement ({self.bad_steps}/{self.patience})")
        print(f"  {'🚨 OVERFIT' if gap > self.gap_threshold else '✅ Gap OK'}")
        print(f"{'='*55}\n")

    def on_train_end(self, args, state, control, **kwargs):
        if not self.history: return
        print("\n" + "="*60)
        print(f"  {'Step':>6}  {'Train':>8}  {'Eval':>8}  {'Gap':>8}  Status")
        print("  " + "-"*54)
        for step, tl, el in self.history:
            gap  = el - tl
            flag = "🚨" if gap > self.gap_threshold else ("⚠️" if gap > 0.2 else "✅")
            print(f"  {step:>6}  {tl:>8.4f}  {el:>8.4f}  {gap:>+8.4f}  {flag}")
        print("="*60)

# ══════════════════════════════════════════════════════════
# 8️⃣  Training Arguments
# 🔧 v3: gradient_accumulation_steps 8 → 4 عشان يعوض الـ batch size مع الـ 2048
# ══════════════════════════════════════════════════════════
training_args = TrainingArguments(
    output_dir="./llama3_8B_lora_v3",
    per_device_train_batch_size=1,
    per_device_eval_batch_size=1,
    gradient_accumulation_steps=4,   # ✅ v3: كان 8 → 4 (عشان max_length أكبر)
    max_steps=500,
    warmup_steps=warmup_steps,
    learning_rate=2e-4,
    weight_decay=0.01,
    lr_scheduler_type="cosine",
    fp16=True,
    logging_steps=20,
    eval_strategy="steps",
    eval_steps=eval_steps,
    save_strategy="steps",
    save_steps=eval_steps,
    save_total_limit=3,
    load_best_model_at_end=True,
    metric_for_best_model="eval_loss",
    greater_is_better=False,
    optim="paged_adamw_8bit",
    report_to="none",
)

# ══════════════════════════════════════════════════════════
# 9️⃣  Train
# ══════════════════════════════════════════════════════════
trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=tokenized_train,
    eval_dataset=tokenized_eval,
    data_collator=DataCollatorForLanguageModeling(tokenizer=tokenizer, mlm=False),
    callbacks=[
        OverfitDetectorCallback(gap_threshold=0.5, patience=3),
        EarlyStoppingCallback(early_stopping_patience=2),
    ],
)

print("\n🚀 Training v3...")
trainer.train()

# ══════════════════════════════════════════════════════════
# 🔟  Final Report & Save
# ══════════════════════════════════════════════════════════
print("\n📊 Final Evaluation:")
print(trainer.evaluate())

logs         = trainer.state.log_history
train_losses = [x["loss"]      for x in logs if "loss" in x and "eval_loss" not in x]
eval_losses  = [x["eval_loss"] for x in logs if "eval_loss" in x]

if train_losses and eval_losses:
    ft, fe = train_losses[-1], eval_losses[-1]
    fg = fe - ft
    print(f"\n🏁 Train={ft:.4f} | Eval={fe:.4f} | Gap={fg:+.4f}")
    if   fg > 0.5: print("   🚨 OVERFITTING")
    elif fg > 0.2: print("   ⚠️  Mild overfitting")
    else:          print("   ✅ No significant overfitting")

trainer.save_model("./llama3_8B_lora_v3")
tokenizer.save_pretrained("./llama3_8B_lora_v3")
print("\n✅ Done — model saved to ./llama3_8B_lora_v3")
