CANDIDATE_K    = 20

# ══════════════════════════════════════════════════════════════════════════════
# TEXT PROCESSING
# ══════════════════════════════════════════════════════════════════════════════

def classify_page(page_num, text):
    if page_num in _TOC_PAGES:      return "toc"
    if page_num in _COVER_PAGES:    return "cover"
    if page_num in _EXERCISE_PAGES: return "exercise"
    for line in [l.strip() for l in text.split("\n") if l.strip()][:4]:
        if re.match(r"^Exercises?$", line, re.I):
            return "exercise"
    return "content"

def clean(text):
    text = re.sub(r"\[Page \d+ of \d+\]\s*", "", text)
    text = re.sub(r"^\s*\d{1,3}\s*$", "", text, flags=re.M)
    text = re.sub(r"^\s*[IVX]{1,5}\s*$", "", text, flags=re.M)
    text = re.sub(r"\r\n", "\n", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    text = re.sub(r"[ \t]{2,}", " ", text)
    return text.strip()

def make_chunks(page_num, text, ptype):
    words = text.split()
    if len(words) <= CHUNK_SIZE:
        return [{"page": page_num, "text": text, "type": ptype}]
    chunks, start = [], 0
    while start < len(words):
        end = min(start + CHUNK_SIZE, len(words))
        chunks.append({"page":page_num,"text":" ".join(words[start:end]),"type":ptype})
        if end == len(words): break
        start += CHUNK_SIZE - CHUNK_OVR
    return chunks


# ══════════════════════════════════════════════════════════════════════════════
# PDF LOADER
# ══════════════════════════════════════════════════════════════════════════════

def load_book(path):
    print(f"  Opening PDF: {path}")
    all_chunks, skipped = [], 0
    with pdfplumber.open(path) as pdf:
        total = len(pdf.pages)
        print(f"  Total pages: {total}")
        for page in pdf.pages:
            pg  = page.page_number
            raw = page.extract_text(layout=True, x_tolerance=3, y_tolerance=3)
            if not raw: skipped += 1; continue
            text = clean(raw)
            if len(text) < MIN_PAGE_CHARS: skipped += 1; continue
            ptype = classify_page(pg, text)
            all_chunks.extend(make_chunks(pg, text, ptype))
    counts = {}
    for c in all_chunks:
        counts[c["type"]] = counts.get(c["type"],0)+1
    print(f"  Pages:{total}  Skipped:{skipped}  Chunks:{len(all_chunks)}  {counts}")
    return all_chunks


# ══════════════════════════════════════════════════════════════════════════════
# STEP 1 — build embeddings BEFORE importing bitsandbytes
#
# ROOT CAUSE of the Kaggle error:
#   sentence-transformers and bitsandbytes both monkey-patch PyTorch ops at
#   import time. When imported in the same process in either order, some ops
#   get registered twice -> RuntimeError: duplicate registrations.
#
# FIX: import sentence-transformers first, encode everything, delete the
#      encoder object and call cuda.empty_cache(). Only THEN import
#      bitsandbytes/transformers for the LLM. The patch conflict disappears
#      because BNB registers its ops into an already-initialised torch state.
# ══════════════════════════════════════════════════════════════════════════════

def build_embedding_index(chunks):
    from sentence_transformers import SentenceTransformer, CrossEncoder
    import torch

    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"  Device: {device}")

    print(f"  Loading embedder ({EMBED_MODEL}) ...")
    embedder = SentenceTransformer(EMBED_MODEL, device=device)
    texts    = [c["text"] for c in chunks]

    print("  Encoding chunks ...")
    sem_mat = embedder.encode(
        texts, batch_size=64, show_progress_bar=True,
        convert_to_numpy=True, normalize_embeddings=True,
    )
    print(f"  Semantic matrix: {sem_mat.shape}")

    print("  Building TF-IDF ...", end=" ", flush=True)
    tfidf     = TfidfVectorizer(ngram_range=(1,2), max_df=0.85,
                                min_df=1, sublinear_tf=True,
                                strip_accents="unicode")
    tfidf_mat = tfidf.fit_transform(texts)
    print(f"done ({tfidf_mat.shape[1]} terms)")

    print(f"  Loading re-ranker ({RERANK_MODEL}) ...")
    reranker = CrossEncoder(RERANK_MODEL, device=device)

    # Free GPU before loading the 4-bit LLM
    del embedder
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
        free_gb = torch.cuda.mem_get_info()[0] / 1e9
        print(f"  GPU cleared. Free: {free_gb:.1f} GB")

    return sem_mat, tfidf, tfidf_mat, reranker


# ══════════════════════════════════════════════════════════════════════════════
# STEP 2 — load LLM after embeddings done + GPU cleared
# ══════════════════════════════════════════════════════════════════════════════

def load_llm():
    import torch
    from transformers import (AutoTokenizer, AutoModelForCausalLM,
                              pipeline, BitsAndBytesConfig)

    print(f"  Loading LLM on {'GPU' if torch.cuda.is_available() else 'CPU'} ...")
    bnb = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_compute_dtype=torch.float16,
        bnb_4bit_use_double_quant=True,
        bnb_4bit_quant_type="nf4",
    )
    tok = AutoTokenizer.from_pretrained(MODEL_ID, token=HF_TOKEN)
    tok.pad_token = tok.eos_token
    mdl = AutoModelForCausalLM.from_pretrained(
        MODEL_ID, token=HF_TOKEN,
        quantization_config=bnb,
        device_map="auto",
        low_cpu_mem_usage=True,
    )
    gen = pipeline("text-generation", model=mdl, tokenizer=tok,
                   pad_token_id=tok.eos_token_id,
                   truncation=True)
    print("  LLM ready!")
    return gen


# ══════════════════════════════════════════════════════════════════════════════
# RETRIEVER
# ══════════════════════════════════════════════════════════════════════════════

class HybridRetriever:
    """
    Two-stage retrieval:
      Stage 1 — hybrid = ALPHA*semantic + (1-ALPHA)*tfidf  (top CANDIDATE_K)
      Stage 2 — cross-encoder re-ranks candidates          (top TOP_K)

    Query encoder is kept on CPU to avoid conflict with the 4-bit LLM on GPU.
    """

    def __init__(self, chunks, sem_mat, tfidf, tfidf_mat, reranker):
        self.chunks    = chunks
        self.sem_mat   = sem_mat
        self.tfidf     = tfidf
        self.tfidf_mat = tfidf_mat
        self.reranker  = reranker
        self._qenc     = None   # lazy-loaded query encoder

    def _get_qencoder(self):
        if self._qenc is None:
            from sentence_transformers import SentenceTransformer
            # CPU only: avoids any GPU state conflict with the 4-bit LLM
            self._qenc = SentenceTransformer(EMBED_MODEL, device="cpu")
        return self._qenc

    @staticmethod
    def _minmax(arr):
        lo, hi = arr.min(), arr.max()
        return (arr - lo) / (hi - lo + 1e-9)

    def retrieve(self, query, top_k=TOP_K, candidate_k=CANDIDATE_K):
        # Semantic score
        q_emb      = self._get_qencoder().encode(
                         [query], normalize_embeddings=True)
        sem_scores = (q_emb @ self.sem_mat.T).flatten()

        # TF-IDF score
        q_tfidf      = self.tfidf.transform([query])
        tfidf_scores = sk_cosine(q_tfidf, self.tfidf_mat).flatten()

        # Hybrid fusion
        hybrid = (ALPHA * self._minmax(sem_scores)
                  + (1 - ALPHA) * self._minmax(tfidf_scores))

        # Page-type penalty
        for i, c in enumerate(self.chunks):
            hybrid[i] *= _PAGE_WEIGHT[c["type"]]

        # One best chunk per page
        candidates, seen = [], set()
        for i in np.argsort(hybrid)[::-1]:
            pg = self.chunks[i]["page"]
            if pg not in seen:
                candidates.append(i)
                seen.add(pg)
            if len(candidates) >= candidate_k:
                break

        # Cross-encoder re-ranking
        pairs   = [(query, self.chunks[i]["text"]) for i in candidates]
        rscores = self.reranker.predict(pairs)
        ranked  = sorted(zip(rscores, candidates),
                         key=lambda x: x[0], reverse=True)

        best = ranked[0][0] if ranked else -999.0
        if best < RERANK_THRESHOLD:
            print(f"  [Not in book — best re-rank:{best:.2f} < {RERANK_THRESHOLD}]")
            return []

        return [
            {"page": self.chunks[i]["page"],
             "text": self.chunks[i]["text"],
             "rerank_score": float(s)}
            for s, i in ranked[:top_k]
        ]


# ══════════════════════════════════════════════════════════════════════════════
# RAG PIPELINE
# ══════════════════════════════════════════════════════════════════════════════

def build_context(hits):
    ordered = sorted(hits, key=lambda h: h["page"])
    return "\n\n".join(
        f"--- Passage {i+1}  (Page {h['page']}) ---\n{h['text']}"
        for i, h in enumerate(ordered)
    )

def ask(query, retriever, gen, history):
    hits = retriever.retrieve(query)
    if not hits:
        return ("This information is not available in the book.\n\n"
                "Source: Not found in book"), []

    context = build_context(hits)
    pages   = sorted({h["page"] for h in hits})
    scores  = [round(h["rerank_score"],2) for h in hits]
    print(f"  [Pages:{pages}  re-rank:{scores}]")

    user_turn = (
        "CONTEXT FROM THE BOOK:\n\n" + context +
        "\n\n" + "-"*50 + "\nQUESTION: " + query
    )
    messages = [{"role":"system","content":SYSTEM_PROMPT}]
    messages += history[-6:]
    messages.append({"role":"user","content":user_turn})

    out    = gen(messages, max_new_tokens=MAX_NEW, temperature=0.1,
                   do_sample=True, truncation=True)
    answer = out[0]["generated_text"][-1]["content"]
    return answer, hits

def main():
    print("\n" + "="*66)
    print("  RAG Book Assistant  -  Database Systems II")
    print("  Retriever : Hybrid Semantic (all-MiniLM-L6-v2) + TF-IDF")
    print("  Re-ranker : Cross-encoder (ms-marco-MiniLM-L-6-v2)")
    print("  Generator : Llama 3.1 8B Instruct (4-bit NF4)")
    print("="*66+"\n")

    print("Step 1/3 - Loading book from PDF ...")
    chunks = load_book(PDF_PATH)
    print()

    print("Step 2/3 - Building retrieval index ...")
    sem_mat, tfidf, tfidf_mat, reranker = build_embedding_index(chunks)
    retriever = HybridRetriever(chunks, sem_mat, tfidf, tfidf_mat, reranker)
    print()

    print("Step 3/3 - Loading language model ...")
    gen = load_llm()
    print()

    history = []
    print("="*66)
    print("  Ready!  Ask anything about the book.")
    print("  Commands:  exit  |  clear")
    print("="*66)

    while True:
        try:
            query = input("\nYou: ").strip()
        except (KeyboardInterrupt, EOFError):
            print("\nGoodbye!"); break

        if not query: continue
        if query.lower() in ("exit","quit"): print("Goodbye!"); break
        if query.lower() == "clear":
            history.clear(); print("Conversation cleared."); continue

        try:
            answer, hits = ask(query, retriever, gen, history)
            print(f"\nAssistant:\n{answer}")
            if hits:
                print(f"\n[Turn {len(history)//2+1} "
                      f"| Pages:{sorted({h['page'] for h in hits})}]")
            history.append({"role":"user","content":query})
            history.append({"role":"assistant","content":answer})
            if len(history) > 12:
                history = history[-12:]

        except Exception as exc:
            import torch
            if "out of memory" in str(exc).lower():
                gc.collect()
                if torch.cuda.is_available(): torch.cuda.empty_cache()
                history.clear()
                print("GPU OOM - history cleared. Try again.")
            else:
                print(f"Error: {exc}")

if __name__ == "__main__":
    main()
